const sha = function (s) { return crypto.createHash('sha256').update(String(s)).digest(); };
const same = function (a, b) { return crypto.timingSafeEqual(sha(a), sha(b)); };
const str = function (v, n) { return typeof v === 'string' ? v.trim().slice(0, n) : ''; };
function sign(p) { const b = Buffer.from(JSON.stringify(p)).toString('base64url'); return b + '.' + crypto.createHmac('sha256', cfg.secret).update(b).digest('base64url'); }
function verify(t) {
  if (typeof t !== 'string') return null; const parts = t.split('.'); if (parts.length !== 2) return null;
  const e = crypto.createHmac('sha256', cfg.secret).update(parts[0]).digest('base64url');
  if (parts[1].length !== e.length || !crypto.timingSafeEqual(Buffer.from(parts[1]), Buffer.from(e))) return null;
  try { const p = JSON.parse(Buffer.from(parts[0], 'base64url').toString()); return p.exp > Date.now() ? p : null; } catch (e) { return null; }
}
const issue = function (p) { p.exp = Date.now() + TOKEN_HOURS * 3600e3; return sign(p); };
function hashPw(pw, saltHex) { return crypto.pbkdf2Sync(String(pw), Buffer.from(saltHex, 'hex'), 60000, 32, 'sha256').toString('hex'); }
const adminHash = function (pw, saltHex) { return crypto.scryptSync(String(pw), Buffer.from(saltHex, 'hex'), 64).toString('hex'); };
function adminOk(username, pw) {
  const a = cfg.admin; if (!a) return false;
  const u = same(String(username || ''), a.username); /* exact match: case, spaces and symbols all count */
  const p = same(adminHash(pw || '', a.salt), a.hash);
  return u && p;
}
function setAdmin(username, pw) {
  const prevName = cfg.admin && cfg.admin.displayName;
  const salt = crypto.randomBytes(16).toString('hex');
  cfg.admin = { username: username, salt: salt, hash: adminHash(pw, salt), ver: ((cfg.admin && cfg.admin.ver) || 0) + 1 };
  if (prevName) cfg.admin.displayName = prevName; /* the editable display name survives a password change */
  fs.writeFileSync(CFG_FILE, JSON.stringify(cfg, null, 2), { mode: 0o600 });
}
function checkNewCreds(username, pw) {
  if (username.length < 3 || username.length > 40 || !/^[A-Za-z0-9._-]+( [A-Za-z0-9._-]+)*$/.test(username)) return 'Username must be 3-40 characters: letters, numbers, dot, dash, underscore, and single spaces between words.';
  if (pw.length < 8 || pw.length > 200) return 'Password needs at least 8 characters.';
  return null;
}
const studentId = function (lrn) { return 's-' + crypto.createHmac('sha256', cfg.secret).update('student:' + lrn).digest('hex').slice(0, 16); }; /* the LRN is never used as the public id */
const adviserOnline = function (section) { return !!presence[section] && Date.now() - presence[section] < 120000; };

/* brute-force guard: only WRONG passwords count. 8 wrong tries per 10 minutes for the same account from the same address. */
const fails = {};
const fkey = function (ip, who) { return ip + '|' + String(who || '').toLowerCase().slice(0, 80); };
function blocked(ip, who) { const k = fkey(ip, who), now = Date.now(); fails[k] = (fails[k] || []).filter(function (t) { return now - t < 600000; }); return fails[k].length >= 8; }
function failed(ip, who) { const k = fkey(ip, who); (fails[k] = fails[k] || []).push(Date.now()); }
function cleared(ip, who) { delete fails[fkey(ip, who)]; }
setInterval(function () { const now = Date.now(); Object.keys(fails).forEach(function (k) { fails[k] = fails[k].filter(function (t) { return now - t < 600000; }); if (!fails[k].length) delete fails[k]; }); }, 600000).unref();
const TOO_MANY = { ok: false, error: 'Too many wrong attempts. Please wait a few minutes and try again.' };

/* ---------- who may see which report (same rules as the website) ---------- */
function canView(s, r) {
  if (!s) return false;
  if (s.role === 'student') return r.reporterRole === 'student' && r.reporterId === s.id;
  if (s.role === 'admin') return r.approval === 'approved' || r.approval === 'none' || (r.approval === 'rejected' && r.rejectedBy === 'admin');
  if (s.role === 'teacher') {
    if (s.advisory && r.reporterRole === 'student' && r.section === s.section) return true;
    if (r.reporterRole === 'teacher' && r.reporterId === s.id) return true;
  }
  return false;
}
function teacherReply(t) { return { ok: true, teacher: { id: t.id, name: t.name, advisory: t.advisory, section: t.section }, token: issue({ role: 'teacher', id: t.id, name: t.name, advisory: t.advisory, section: t.section }) }; }
const awaitingCount = function () { return db.reports.filter(function (r) { return r.approval === 'pending'; }).length; };
const visible = function (s) { return s ? db.reports.filter(function (r) { return canView(s, r); }) : []; };

/* ---------- own profile: a person can only ever read or change THEIR OWN account (the account is taken from the login token, never from the request) ---------- */
function profileOf(s) {
  if (!s) return null;
  if (s.role === 'admin') return cfg.admin ? { role: 'admin', id: 'u-admin', name: cfg.admin.displayName || cfg.admin.username, username: cfg.admin.username, custom: !!cfg.admin.displayName } : null;
  if (s.role === 'teacher') { const t = db.teachers.filter(function (x) { return x.id === s.id; })[0]; return t ? { role: 'teacher', id: t.id, username: t.username, name: t.name, advisory: !!t.advisory, section: t.section || null } : null; }
  if (s.role === 'student') { const u = db.students.filter(function (x) { return x.id === s.id; })[0]; return u ? { role: 'student', id: u.id, lrn: u.lrn, name: u.name, grade: u.grade, section: u.section, adviser: u.adviser } : null; }
  return null;
}

/* ---------- report writes: the server decides what each person is allowed to change ---------- */
const MUT = ['stage', 'approval', 'approvalBy', 'approvalAt', 'approvalNote', 'rejectedBy', 'delayed', 'delayFixedAt', 'accomplishedAt'];
function cleanHist(h) {
  return Array.isArray(h) ? h.slice(0, 200).filter(function (x) { return x && typeof x === 'object'; }).map(function (x) {
    return { at: +x.at || Date.now(), by: str(x.by, 120), role: str(x.role, 20), action: str(x.action, 200), detail: str(x.detail, 1000), stage: +x.stage || 0 };
  }) : [];
}
function nextId() {
  let n = +db.seq || 0; db.reports.forEach(function (r) { const m = /^RP-(\d+)$/.exec(r.id); if (m && +m[1] > n) n = +m[1]; });
  n++; db.seq = n; return 'RP-' + (n < 10 ? '0' + n : n);
}
function savePhoto(id, dataUrl) { /* a report photo: JPEG only, small, saved as data/photos/RP-xx.jpg */
  const m = /^data:image\/jpeg;base64,([A-Za-z0-9+\/=]+)$/.exec(typeof dataUrl === 'string' && dataUrl.length < 600000 ? dataUrl : '');
  if (!m) return false;
  const buf = Buffer.from(m[1], 'base64');
  if (buf.length < 100 || buf[0] !== 0xFF || buf[1] !== 0xD8) return false; /* must really be a JPEG */
  try { fs.writeFileSync(path.join(PHOTO_DIR, id + '.jpg'), buf); return true; } catch (e) { return false; }
}
function createReport(s, r) {
  if (s.role !== 'student' && s.role !== 'teacher') return;
  const n = { title: str(r.title, 200), description: str(r.description, 3000), location: str(r.location, 300), category: str(r.category, 100) };
  if (!n.title || !n.description) return;
  if (s.role === 'student') {
    if (r.reporterRole !== 'student' || r.reporterId !== s.id || !adviserOnline(s.section)) return;
    Object.assign(n, { reporterId: s.id, reporterName: s.name, reporterRole: 'student', section: s.section, adviserName: s.adviser, stage: 2, approval: 'pending' });
  } else {
    if (r.reporterRole !== 'teacher' || r.reporterId !== s.id) return;
    Object.assign(n, { reporterId: s.id, reporterName: s.name, reporterRole: 'teacher', section: null, adviserName: null, stage: 3, approval: 'none' });
  }
  const now = Date.now();
  n.delayed = false; n.createdAt = +r.createdAt || now; n.updatedAt = now; n.updatedBy = s.name; n.history = cleanHist(r.history);
  const id = str(r.id, 20);
  n.id = /^RP-\d+$/.test(id) && !db.reports.some(function (x) { return x.id === id; }) ? id : nextId();
  const m = /^RP-(\d+)$/.exec(n.id); if (m && +m[1] > db.seq) db.seq = +m[1];
  if (r.photo && savePhoto(n.id, r.photo)) n.hasPhoto = true;
  db.reports.push(n);
}
function cleanField(k, v, old) {
  if (k === 'stage') return Number.isInteger(v) && v >= 1 && v <= 6 ? v : old;
  if (k === 'approval') return ['pending', 'approved', 'rejected', 'none'].indexOf(v) > -1 ? v : old;
  if (k === 'delayed') return !!v;
  if (k === 'approvalBy') return str(v, 120);
  if (k === 'approvalNote') return str(v, 1000);
  if (k === 'rejectedBy') return str(v, 20);
  return typeof v === 'number' && isFinite(v) ? v : null; /* approvalAt, delayFixedAt, accomplishedAt */
}
function updateReport(s, ex, r) {
  if (!canView(s, ex)) return;
  const ih = cleanHist(r.history), eh = ex.history || [];
  if (ih.length <= eh.length) return; /* every real change adds a history line; anything else is stale or empty */
  for (let i = 0; i < eh.length; i++) if (JSON.stringify(ih[i]) !== JSON.stringify(eh[i])) return;
  const diff = MUT.filter(function (k) { return JSON.stringify(r[k]) !== JSON.stringify(ex[k]); });
  let ok = false;
  if (s.role === 'admin') ok = true;
  else if (s.role === 'teacher' && s.advisory && ex.reporterRole === 'student' && ex.section === s.section && ex.approval === 'pending') {
    ok = (r.approval === 'approved' || r.approval === 'rejected') && (r.stage === ex.stage || r.stage === 3) &&
      diff.every(function (k) { return ['approval', 'approvalBy', 'approvalAt', 'approvalNote', 'stage', 'delayed', 'delayFixedAt'].indexOf(k) > -1; });
  }
  if (!ok) return;
  MUT.forEach(function (k) { if (k in r) ex[k] = cleanField(k, r[k], ex[k]); else delete ex[k]; });
  ex.history = ih; ex.updatedAt = Date.now(); ex.updatedBy = s.name;
}
function applyReports(s, list) {
  list.slice(0, 500).forEach(function (r) {
    if (!r || typeof r !== 'object') return;
    const ex = db.reports.filter(function (x) { return x.id === r.id; })[0];
    if (ex && ex.reporterId === r.reporterId && ex.createdAt === +r.createdAt) updateReport(s, ex, r);
    else createReport(s, r);
  });
  save();
}

/* ---------- homepage content ---------- */
function validContent(c) {
  if (!c || typeof c !== 'object') return null;
  const out = { seeded: 1 };
  for (const t of ['grand', 'latest', 'events']) {
    if (!Array.isArray(c[t]) || c[t].length > 200) return null;
    out[t] = c[t].map(function (x) {
      const o = { id: str(x && x.id, 40) || ('c' + Date.now()), title: str(x && x.title, 200), text: str(x && x.text, 1000) };
      if (x && typeof x.desc === 'string' && x.desc.trim()) o.desc = str(x.desc, 2000);
      if (x && /^\d{4}-\d{2}-\d{2}$/.test(x.date || '')) o.date = x.date;
      if (x && typeof x.img === 'string' && x.img.length < 700000 && /^data:image\/(jpeg|png|webp);base64,[A-Za-z0-9+\/=]+$/.test(x.img)) o.img = x.img;
      return o;
    }).filter(function (o) { return o.title; });
  }
  if (c.portal && typeof c.portal === 'object') out.portal = validPortal(c.portal);
  return out;
}
/* ASHS Portal content (Academic Guide sections, SBMS link, Persons In Charge) edited by the admin */
function validPortal(p) {
  const arr = function (a, max) { return Array.isArray(a) ? a.slice(0, max) : []; };
  const url = str(p.sbms && p.sbms.url, 500);
  const out = { sbms: { url: /^https?:\/\/[^\s<>"]+$/i.test(url) ? url : '' } };
  out.guide = arr(p.guide, 30).filter(function (s) { return s && typeof s === 'object'; }).map(function (s) {
    const o = { id: str(s.id, 40) || ('s' + Date.now()), type: s.type === 'buildings' ? 'buildings' : 'steps', title: str(s.title, 100), heading: str(s.heading, 150), intro: str(s.intro, 1000) };
    if (o.type === 'buildings') {
      o.search = str(s.search, 100);
      o.buildings = arr(s.buildings, 20).filter(function (b) { return b && typeof b === 'object'; }).map(function (b) {
        return { name: str(b.name, 100), floors: arr(b.floors, 30).filter(function (f) { return f && typeof f === 'object'; }).map(function (f) {
          return { label: str(f.label, 60), rooms: arr(f.rooms, 80).filter(function (r) { return r && typeof r === 'object'; }).map(function (r) { return { tag: str(r.tag, 30), name: str(r.name, 100) }; }).filter(function (r) { return r.name; }) };
        }) };
      });
    } else {
      o.cards = arr(s.cards, 12).filter(function (c2) { return c2 && typeof c2 === 'object'; }).map(function (c2) {
        return { title: str(c2.title, 150), steps: arr(c2.steps, 40).map(function (x) { return str(x, 500); }).filter(Boolean), noteTitle: str(c2.noteTitle, 100), note: str(c2.note, 1000) };
      });
      o.foot = str(s.foot, 600);
    }
    return o;
  }).filter(function (s) { return s.title; });
  const pfile = function (v) { v = str(v, 80); return /^[a-f0-9]{24}\.(png|jpg|pdf)$/.test(v) ? v : ''; };
  out.persons = arr(p.persons, 150).filter(function (x) { return x && typeof x === 'object'; }).map(function (x) {
    const lv = Math.round(+x.level);
    const o = { id: str(x.id, 40) || ('p' + Date.now()), name: str(x.name, 100), role: str(x.role, 150), level: lv >= 1 && lv <= 6 ? lv : 1 };
    if (typeof x.parent === 'string') o.parent = str(x.parent, 40); /* id of the person this one reports to ('' = top of the chart) */
    if (typeof x.dept === 'string' && x.dept.trim()) o.dept = str(x.dept, 100);
    const ph = pfile(x.photo); if (ph && /\.jpg$/.test(ph)) o.photo = ph;
    return o;
  }).filter(function (x) { return x.name; });
  const ids = {}; out.persons.forEach(function (x) { ids[x.id] = x; });
  out.persons.forEach(function (x) { if (x.parent !== undefined && (x.parent === x.id || !ids[x.parent])) x.parent = ''; }); /* a parent must exist and cannot be the person themself */
  out.persons.forEach(function (x) { let c = x, n = 0; while (c.parent && n++ < 500) { if (c.parent === x.id) { x.parent = ''; break; } c = ids[c.parent]; } }); /* no loops: nobody may report to their own subordinate */
  /* organizational chart settings: which version the public page shows + the uploaded chart file. Switching never deletes either one. */
  const org = p.org && typeof p.org === 'object' ? p.org : {};
  const uf = pfile(org.file);
  out.org = { mode: org.mode === 'upload' && uf ? 'upload' : 'manual' };
  if (uf) out.org.file = uf; if (uf) out.org.kind = /\.pdf$/.test(uf) ? 'pdf' : 'image'; if (uf && str(org.name, 120)) out.org.name = str(org.name, 120);
  return out;
}

const TYPES = { '.html': 'text/html; charset=utf-8', '.css': 'text/css; charset=utf-8', '.js': 'text/javascript; charset=utf-8', '.json': 'application/json', '.png': 'image/png', '.jpg': 'image/jpeg', '.jpeg': 'image/jpeg', '.svg': 'image/svg+xml', '.ico': 'image/x-icon', '.webp': 'image/webp', '.txt': 'text/plain; charset=utf-8' };
function send(res, code, obj) { const b = JSON.stringify(obj); res.writeHead(code, { 'Content-Type': 'application/json; charset=utf-8', 'Cache-Control': 'no-store' }); res.end(b); }
function body(req, limit) {
  return new Promise(function (ok, bad) {
    let n = 0; const chunks = [];
    req.on('data', function (c) { n += c.length; if (n > limit) { bad(new Error('too big')); req.destroy(); } else chunks.push(c); });
    req.on('end', function () { try { ok(JSON.parse(Buffer.concat(chunks).toString() || '{}')); } catch (e) { bad(new Error('bad json')); } });
    req.on('error', bad);
  });
}

    if (!c) return send(res, 400, { ok: false, error: 'Bad content.' });
    db.content = c; db.contentRev = (+db.contentRev || 0) + 1; save();
    return send(res, 200, { ok: true, rev: db.contentRev, content: c });
  }
  return send(res, 404, { ok: false, error: 'Not found.' });
}

function serveStatic(req, res, url) {
  let p = decodeURIComponent(url.pathname); if (p.endsWith('/')) p += 'index.html';
  const f = path.normalize(path.join(PUB, p));
  if (f !== PUB && !f.startsWith(PUB + path.sep)) { res.writeHead(403); return res.end('Forbidden'); }
  fs.stat(f, function (e, st) {
    if (e || !st.isFile()) { res.writeHead(404, { 'Content-Type': 'text/plain' }); return res.end('Not found'); }
    const ext = path.extname(f).toLowerCase();
    res.writeHead(200, { 'Content-Type': TYPES[ext] || 'application/octet-stream', 'Cache-Control': 'no-cache', 'X-Content-Type-Options': 'nosniff' });
    if (req.method === 'HEAD') return res.end();
    fs.createReadStream(f).pipe(res);
  });
}

    api(req, res, url).catch(function (e) { if (!res.headersSent) send(res, e && e.message === 'too big' ? 413 : 400, { ok: false, error: 'Bad request.' }); });
